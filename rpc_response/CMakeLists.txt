cmake_minimum_required(VERSION 3.16)
project(rpc_response LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rpc_response INTERFACE)
target_include_directories(rpc_response INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(rpc_validate tests/validate.cc)
target_link_libraries(rpc_validate PRIVATE rpc_response)
add_executable(rpc_muon_waveform_reference scripts/muon_waveform_reference.cc)
target_link_libraries(rpc_muon_waveform_reference PRIVATE rpc_response)
enable_testing()
add_test(NAME rpc_physics_validation COMMAND rpc_validate)
option(RPC_WITH_ROOT "Build ROOT segment digitizer" OFF)
option(RPC_WITH_GARFIELD "Build Garfield++ transport exporter (requires ROOT)" OFF)
if(RPC_WITH_ROOT OR RPC_WITH_GARFIELD)
  find_package(ROOT REQUIRED COMPONENTS Core RIO Tree TreePlayer Hist)
endif()
if(RPC_WITH_ROOT)
  add_executable(rpc_digitize src/digitize.cc)
  target_link_libraries(rpc_digitize PRIVATE rpc_response ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
  add_executable(rpc_root_check tests/root_check.cc)
  target_link_libraries(rpc_root_check PRIVATE rpc_response ROOT::Core ROOT::RIO ROOT::Tree)
  add_executable(rpc_make_kernel src/make_kernel.cc)
  target_link_libraries(rpc_make_kernel PRIVATE rpc_response ROOT::Core ROOT::RIO ROOT::Tree)
  add_executable(rpc_query_kernel src/query_kernel.cc)
  target_link_libraries(rpc_query_kernel PRIVATE rpc_response ROOT::Core ROOT::RIO ROOT::Tree)
  add_executable(rpc_export_comparison src/export_comparison.cc)
  target_link_libraries(rpc_export_comparison PRIVATE rpc_response ROOT::Core ROOT::RIO ROOT::Tree)
  ROOT_GENERATE_DICTIONARY(RpcLegacyDictionary
    ${CMAKE_CURRENT_SOURCE_DIR}/../include/Object.hh
    LINKDEF ${CMAKE_CURRENT_SOURCE_DIR}/../src/ObjectLinkDef.h
    OPTIONS -I${CMAKE_CURRENT_SOURCE_DIR}/../include)
  add_executable(rpc_run_legacy_analysis src/run_legacy_analysis.cc ${CMAKE_CURRENT_BINARY_DIR}/RpcLegacyDictionary.cxx)
  target_include_directories(rpc_run_legacy_analysis PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../include)
  target_link_libraries(rpc_run_legacy_analysis PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist)
endif()
if(RPC_WITH_GARFIELD)
  find_package(Garfield REQUIRED)
  add_executable(rpc_make_response src/make_response.cc)
  target_link_libraries(rpc_make_response PRIVATE rpc_response Garfield::Garfield ROOT::Core ROOT::RIO ROOT::Tree)
  add_executable(rpc_make_gas src/make_gas.cc)
  target_link_libraries(rpc_make_gas PRIVATE rpc_response Garfield::Garfield)
  add_executable(rpc_make_muon_standard src/make_muon_standard.cc)
  target_link_libraries(rpc_make_muon_standard PRIVATE rpc_response Garfield::Garfield ROOT::Core ROOT::RIO ROOT::Tree)
endif()
